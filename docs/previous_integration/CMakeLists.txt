cmake_minimum_required(VERSION 3.20)
option(ROBO_TARGET "Build Pico W with micro T-Kernel" OFF)
option(ROBO_MANUAL_CONSOLE "Deliberately enable USB bring-up console in WiFi task slot" OFF)
if(ROBO_TARGET)
    set(PICO_BOARD pico_w CACHE STRING "Board")
    if(NOT PICO_SDK_PATH)
        set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
    endif()
    include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
endif()
project(smart_robo C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(motion_core src/control/math.c src/control/motion.c
            src/config/board_config.c src/app/vehicle_controller.c)
target_include_directories(motion_core PUBLIC include)
if(NOT MSVC)
    target_compile_options(motion_core PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
endif()
if(ROBO_TARGET)
    add_compile_definitions(PICO_DIVIDER_DISABLE_INTERRUPTS=1)
    set(PICO_DEFAULT_THREAD_LOCAL_IMPL global)
    pico_sdk_init()
    include(cmake/micro_tkernel.cmake)
    add_executable(smart_robo src/app/tasks.c src/app/manual_console.c src/hal/pico_motion.c
                   src/subsystems/placeholders.c src/rtos/pico_startup.c)
    target_link_libraries(smart_robo motion_core micro_tkernel pico_stdlib hardware_pwm hardware_clocks hardware_sync m)
    target_compile_options(smart_robo PRIVATE -Wall -Wextra)
    target_compile_definitions(smart_robo PRIVATE PICO_STACK_SIZE=4096)
    if(ROBO_MANUAL_CONSOLE)
        target_compile_definitions(smart_robo PRIVATE ROBO_MANUAL_CONSOLE=1)
        pico_enable_stdio_usb(smart_robo 1)
    else()
        pico_enable_stdio_usb(smart_robo 0)
    endif()
    pico_enable_stdio_uart(smart_robo 0)
    pico_add_extra_outputs(smart_robo)
else()
    enable_testing()
    add_executable(motion_tests tests/test_motion.c)
    target_link_libraries(motion_tests motion_core)
    if(NOT MSVC)
        target_link_libraries(motion_tests m)
        target_compile_options(motion_tests PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME motion_tests COMMAND motion_tests)
endif()
